TypeScript — Zod Schema

import { z } from "zod";

// Define a message schema with discriminated union
const MessageSchema = z.discriminatedUnion("type", [
  z.object({
    type: z.literal("GET_PAGE_INFO"),
  }),
  z.object({
    type: z.literal("PAGE_INFO"),
    payload: z.object({
      hostname: z.string(),
      url: z.string().url(),
      title: z.string(),
    }),
  }),
]);

type Message = z.infer<typeof MessageSchema>;

function parseMessage(raw: unknown): Message | null {
  const result = MessageSchema.safeParse(raw);
  return result.success ? result.data : null;
}

Python — Data Processing

Python 
from dataclasses import dataclass, field
from typing import Optional, List

# Configuration dataclass with validation
@dataclass
class ReadingConfig:
    font_size: int = 16
    line_height: float = 1.6
    theme: str = "light"
    max_width: int = 780
    site_overrides: dict = field(default_factory=dict)

    def __post_init__(self):
        assert 10 <= self.font_size <= 48, f"font_size out of range: {self.font_size}"
        assert self.theme in ("light", "dark", "sepia")

    def resolve_for_site(self, hostname: str) -> "ReadingConfig":
        """Merge site override on top of global config."""
        overrides = self.site_overrides.get(hostname, {})
        return ReadingConfig(**{**self.__dict__, **overrides})

Rust — Safe String Handling

Rust 
use std::collections::HashMap;

/// A registry mapping operation IDs to CSS properties.
#[derive(Debug, Clone)]
pub struct OperationRegistry {
    entries: HashMap<String, String>,
}

impl OperationRegistry {
    pub fn new() -> Self {
        Self { entries: HashMap::new() }
    }

    pub fn register(&mut self, id: &str, css: String) {
        self.entries.insert(id.to_string(), css);
    }

    pub fn get(&self, id: &str) -> Option<&String> {
        self.entries.get(id)
    }
}

Note on code preservation

The Adaptive Web Layer must never alter font, color, line-height, or letter-spacing inside pre, code, and kbd elements. These are in the protected set of every Operation.
